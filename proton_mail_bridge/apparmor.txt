#include <tunables/global>

profile proton_mail_bridge flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  file,
  network,
  capability,
  signal (send) set=(kill,term,int,hup,cont),

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /usr/local/bin/** ix,
  /usr/sbin/** ix,
  /sbin/** ix,
  /package/** ix,
  /command/** ix,
  /etc/s6-overlay/** rwix,
  /run/{s6,s6-rc*,service}/** ix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio & temp
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Storage & devices
  /data/** rw,
  /ssl/** rw,
  /dev/null rw,
  /dev/zero rw,
  /dev/urandom r,
  /dev/ptmx rw,
  /dev/pts/** rw,

  # Deny sensitive host paths
  deny /proc/sysrq-trigger rwklx,
}
